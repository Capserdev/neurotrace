def load_folder_rgb(path, label, target_size=(IMG_SIZE, IMG_SIZE)):
    """Load all images in path as float32 RGB (0..255)."""
    xs, ys = [], []
    if not os.path.exists(path):
        return xs, ys
    files = sorted(os.listdir(path))
    for f in files:
        if f.lower().endswith((".png", ".jpg", ".jpeg")):
            p = os.path.join(path, f)
            try:
                im = load_img(p, color_mode="rgb", target_size=target_size)
                a = img_to_array(im).astype("float32")  # 0..255
                xs.append(a)
                ys.append(label)
            except Exception:
                pass
    return xs, ys

def summarize_labels(y, name):
    c = Counter(y)

def build_effnet_b0_model(name="EffNetV2B0_Model"):
    base = EfficientNetV2B0(
        include_top=False,
        weights="imagenet",
        input_shape=(IMG_SIZE, IMG_SIZE, 3),
    )
    base.trainable = False  # start frozen

    inputs = layers.Input(shape=(IMG_SIZE, IMG_SIZE, 3), name="img")
    x = preprocess_input(inputs)
    x = base(x, training=False)
    x = layers.GlobalAveragePooling2D()(x)
    x = layers.Dropout(0.4)(x)
    x = layers.Dense(
        256,
        activation="relu",
        kernel_regularizer=tf.keras.regularizers.l2(1e-4),
    )(x)
    x = layers.Dropout(0.4)(x)
    outputs = layers.Dense(1, activation="sigmoid")(x)
    model = models.Model(inputs, outputs, name=name)
    return model, base

def train_head_and_finetune(
    model,
    base,
    X_tr,
    y_tr,
    X_val,
    y_val,
    class_weight,
    model_prefix,
    epochs_head=EPOCHS_HEAD,
    epochs_fine=EPOCHS_FINE,
):
   
    base.trainable = False
    model.compile(
        optimizer=optimizers.Adam(1e-3),
        loss="binary_crossentropy",
        metrics=["accuracy"],
    )
    ckpt_head = f"/kaggle/working/{model_prefix}_head_best.keras"
    es = callbacks.EarlyStopping(
        monitor="val_loss", patience=4, restore_best_weights=True, verbose=1
    )
    rl = callbacks.ReduceLROnPlateau(
        monitor="val_loss", factor=0.3, patience=2, verbose=1
    )
    mc = callbacks.ModelCheckpoint(
        ckpt_head, monitor="val_loss", save_best_only=True, verbose=1
    )

    print(f"\n=== HEAD TRAINING for {model_prefix} ===")
    hist_head = model.fit(
        X_tr,
        y_tr,
        validation_data=(X_val, y_val),
        epochs=epochs_head,
        batch_size=BATCH,
        class_weight=class_weight,
        callbacks=[es, rl, mc],
        verbose=2,
    )

    
    print(f"\nUnfreezing backbone for {model_prefix} ...")
    base.trainable = True
    for layer in base.layers:
        if isinstance(layer, layers.BatchNormalization):
            layer.trainable = False

    model.compile(
        optimizer=optimizers.Adam(1e-5),
        loss="binary_crossentropy",
        metrics=["accuracy"],
    )
    ckpt_fine = f"/kaggle/working/{model_prefix}_finetune_best.keras"
    es2 = callbacks.EarlyStopping(
        monitor="val_loss", patience=4, restore_best_weights=True, verbose=1
    )
    rl2 = callbacks.ReduceLROnPlateau(
        monitor="val_loss", factor=0.3, patience=2, verbose=1
    )
    mc2 = callbacks.ModelCheckpoint(
        ckpt_fine, monitor="val_loss", save_best_only=True, verbose=1
    )

    print(f"\n=== FINE-TUNING for {model_prefix} ===")
    hist_fine = model.fit(
        X_tr,
        y_tr,
        validation_data=(X_val, y_val),
        epochs=epochs_fine,
        batch_size=BATCH,
        class_weight=class_weight,
        callbacks=[es2, rl2, mc2],
        verbose=2,
    )

    if os.path.exists(ckpt_fine):
        model = models.load_model(ckpt_fine)
    elif os.path.exists(ckpt_head):
        model = models.load_model(ckpt_head)

    return model, hist_head, hist_fine

def eval_model(model, X, y, name=""):
    if len(y) == 0:
        print(f"{name}: NO DATA")
        return
    preds_prob = model.predict(X, batch_size=BATCH, verbose=0).ravel()
    preds = (preds_prob > 0.5).astype(int)
    acc = accuracy_score(y, preds)
    print(f"\n{name} accuracy: {acc*100:.2f}%   (N={len(y)})")
    print(classification_report(y, preds, target_names=["Healthy","Parkinson"]))
    cm = confusion_matrix(y, preds)
    plt.figure(figsize=(4,3))
    sns.heatmap(
        cm, annot=True, fmt="d", cmap="Blues",
        xticklabels=["H","P"], yticklabels=["H","P"]
    )
    plt.title(f"{name} Confusion Matrix")
    plt.xlabel("Predicted")
    plt.ylabel("True")
    plt.tight_layout()
    plt.show()
    try:
        fpr, tpr, _ = roc_curve(y, preds_prob)
        auc = roc_auc_score(y, preds_prob)
        plt.figure(figsize=(4,3))
        plt.plot(fpr, tpr, label=f"AUC={auc:.3f}")
        plt.plot([0,1],[0,1],'k--')
        plt.title(f"{name} ROC")
        plt.xlabel("FPR")
        plt.ylabel("TPR")
        plt.legend()
        plt.tight_layout()
        plt.show()
    except Exception as e:
        print(f"{name} ROC error:", e)

def tta_probs(model, X, batch_size=BATCH):
    """TTA: original + hflip + 90° + 270°."""
    if len(X) == 0:
        return np.zeros((0,), dtype="float32")

    preds = []

    # original
    preds.append(model.predict(X, batch_size=batch_size, verbose=0).ravel())

    # horizontal flip
    X_hf = X[:, :, ::-1, :]
    preds.append(model.predict(X_hf, batch_size=batch_size, verbose=0).ravel())

    # rotate 90° ccw
    X_r1 = np.rot90(X, k=1, axes=(1, 2))
    preds.append(model.predict(X_r1, batch_size=batch_size, verbose=0).ravel())

    # rotate 90° cw
    X_r3 = np.rot90(X, k=3, axes=(1, 2))
    preds.append(model.predict(X_r3, batch_size=batch_size, verbose=0).ravel())

    return np.mean(preds, axis=0)

def eval_ensemble_with_threshold_and_tta(
    models, X_val, y_val, X_tests, y_tests, names
):
    
    val_probs_list = [tta_probs(m, X_val) for m in models]
    val_probs = np.mean(val_probs_list, axis=0)

    thresholds = np.linspace(0.2, 0.8, 61)
    best_thr = 0.5
    best_acc = 0.0
    for thr in thresholds:
        preds = (val_probs > thr).astype(int)
        acc = accuracy_score(y_val, preds)
        if acc > best_acc:
            best_acc = acc
            best_thr = thr
    print(f"\n[Spiral ENSEMBLE] Best val-threshold = {best_thr:.3f} with val accuracy = {best_acc*100:.2f}%")

    
    for X_test, y_test, name in zip(X_tests, y_tests, names):
        if len(y_test) == 0:
            print(f"{name}: NO DATA")
            continue
        probs_list = [tta_probs(m, X_test) for m in models]
        probs = np.mean(probs_list, axis=0)
        preds = (probs > best_thr).astype(int)
        acc = accuracy_score(y_test, preds)
        print(f"\n{name} (ENSEMBLE thr={best_thr:.3f} + TTA) accuracy: {acc*100:.2f}%   (N={len(y_test)})")
        print(classification_report(y_test, preds, target_names=["Healthy","Parkinson"]))
        cm = confusion_matrix(y_test, preds)
        plt.figure(figsize=(4,3))
        sns.heatmap(
            cm, annot=True, fmt="d", cmap="Blues",
            xticklabels=["H","P"], yticklabels=["H","P"]
        )
        plt.title(f"{name} Confusion Matrix (Ensemble)")
        plt.xlabel("Predicted")
        plt.ylabel("True")
        plt.tight_layout()
        plt.show()

        try:
            fpr, tpr, _ = roc_curve(y_test, probs)
            auc = roc_auc_score(y_test, probs)
            plt.figure(figsize=(4,3))
            plt.plot(fpr, tpr, label=f"AUC={auc:.3f}")
            plt.plot([0,1],[0,1],'k--')
            plt.title(f"{name} ROC (Ensemble)")
            plt.xlabel("FPR")
            plt.ylabel("TPR")
            plt.legend()
            plt.tight_layout()
            plt.show()
        except Exception as e:
            print(f"{name} ROC error:", e)

def build_effnet_b0_wave(name="WaveModel"):
    base = EfficientNetV2B0(
        include_top=False,

def tta_probs(model, X, batch_size=BATCH):
    """TTA: original + hflip + 90° + 270°."""
    if len(X) == 0:
        return np.zeros((0,), dtype="float32")

    preds = []

    
    preds.append(model.predict(X, batch_size=batch_size, verbose=0).ravel())

   
    X_hf = X[:, :, ::-1, :]
    preds.append(model.predict(X_hf, batch_size=batch_size, verbose=0).ravel())

  
    X_r1 = np.rot90(X, k=1, axes=(1, 2))
    preds.append(model.predict(X_r1, batch_size=batch_size, verbose=0).ravel())

   
    X_r3 = np.rot90(X, k=3, axes=(1, 2))
    preds.append(model.predict(X_r3, batch_size=batch_size, verbose=0).ravel())

    return np.mean(preds, axis=0)

def extract_features_for_file(path_str: str):
    """
    Load an audio file and extract a fixed-length feature vector.
    - Skips files shorter than 1 second
    - For text ('ReadText' in path), skips the first 30 seconds.
    Returns:
        list of 36 floats, or None if something goes wrong / too short.
    """
    path_str = str(path_str)

    # 1) Load audio
    try:
        y, sr = librosa.load(path_str, sr=None, mono=True)
    except Exception as e:
        print(f"[SKIP] {path_str}: {e}")
        return None

    # 2) Skip very short files globally (<1s)
    dur = len(y) / float(sr)
    if dur < 1.0:
        print(f"[SKIP <1s] {path_str}: duration={dur:.3f}s")
        return None

    # 3) For text recordings, skip first 30 seconds
    norm_path = path_str.replace("\\", "/")
    if "ReadText" in norm_path:
        offset_samples = min(len(y), int(33.0 * sr))
        y = y[offset_samples:]

        # Re-check duration after skipping 30s
        dur_after = len(y) / float(sr)
        if dur_after < 1.0:
            print(f"[SKIP text <1s after 30s offset] {path_str}: duration={dur_after:.3f}s")
            return None
    else:
        dur_after = len(y) / float(sr)

    # 4) Feature extraction
    feats = []

    # Duration (seconds) after processing
    feats.append(dur_after)

    # RMS energy
    try:
        rms = librosa.feature.rms(y=y)[0]
        feats.extend([rms.mean(), rms.std()])
    except Exception:
        feats.extend([0.0, 0.0])

    # Zero-crossing rate
    try:
        zcr = librosa.feature.zero_crossing_rate(y)[0]
        feats.extend([zcr.mean(), zcr.std()])
    except Exception:
        feats.extend([0.0, 0.0])

    # Pitch (f0) via librosa.yin
    try:
        f0 = librosa.yin(y, fmin=50, fmax=500, sr=sr)
        feats.extend([np.nanmean(f0), np.nanstd(f0)])
    except Exception:
        feats.extend([0.0, 0.0])

    # MFCCs (13 coefficients → mean + std = 26 features)
    try:
        mfcc = librosa.feature.mfcc(y=y, sr=sr, n_mfcc=13)
        mfcc_mean = mfcc.mean(axis=1)
        mfcc_std = mfcc.std(axis=1)
        feats.extend(mfcc_mean.tolist())
        feats.extend(mfcc_std.tolist())
    except Exception:
        # If MFCC fails, pad with zeros
        feats.extend([0.0] * 26)

    # 5) Ensure fixed length of 36 numeric features
    if len(feats) < 36:
        feats = feats + [0.0] * (36 - len(feats))
    else:
        feats = feats[:36]

    return feats


def build_dataframe_from_files(file_paths, labels):
    """
    Given a list of paths and labels, extract features for each file
    and build a DataFrame with:
      - 'path'
      - 'label'
      - f0 ... f35
    Files where feature extraction fails / is skipped are omitted.
    """
    rows = []
    iterable = list(zip(file_paths, labels))
    for p, lab in tqdm(iterable, desc="Extracting features", total=len(iterable)):
        feats = extract_features_for_file(str(p))
        if feats is None:
            continue

        row = {"path": str(p), "label": lab}
        for i, v in enumerate(feats):
            row[f"f{i}"] = v
        rows.append(row)

    df = pd.DataFrame(rows)
    print("Feature table shape:", df.shape)
    print(df["label"].value_counts())
    return df

from sklearn.model_selection import train_test_split, StratifiedKFold, GridSearchCV
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.svm import SVC
from sklearn.metrics import classification_report, confusion_matrix, accuracy_score
import numpy as np
import joblib

def evaluate_multiclass_model(
    df,
    model_name="multiclass_model",
    test_size=0.2,
    random_state=42,
    save_path="/kaggle/working"
):
    """
    Train a 4-class SVM model using:
      - StandardScaler
      - RBF SVC(probability=True, decision_function_shape='ovr')
      - StratifiedKFold CV
      - Larger search space over C and gamma
      - class_weight='balanced' for skew handling

    Assumes df has:
      - 'label': class strings (e.g., 'HC', 'PD', 'PSP', 'MSA')
      - feature columns starting with "f"
    """

    # -------------------------
    # 1) Features / labels
    # -------------------------
    feature_cols = [c for c in df.columns if c.startswith("f")]
    print(f"[{model_name}] Using {len(feature_cols)} feature columns")

    X = df[feature_cols].values
    y_labels = df["label"].values

    # Encode labels as integers for the classifier
    classes = sorted(np.unique(y_labels))
    class_to_int = {cls: i for i, cls in enumerate(classes)}
    int_to_class = {v: k for k, v in class_to_int.items()}
    y = np.array([class_to_int[c] for c in y_labels])

    print("Label counts:")
    print(df["label"].value_counts())

    

    # -------------------------
    # 2) Train / test split
    # -------------------------
    X_train, X_test, y_train, y_test = train_test_split(
        X,
        y,
        test_size=test_size,
        random_state=random_state,
        stratify=y,
    )

    print("Train size:", X_train.shape[0], ", Test size:", X_test.shape[0])

    # -------------------------
    # 3) Pipeline + param grid
    # -------------------------
    pipe = Pipeline([
        ("sc", StandardScaler()),
        (
            "clf",
            SVC(
                kernel="rbf",
                probability=True,
                decision_function_shape="ovr",
                class_weight="balanced",   # <-- helps with slight imbalances
                random_state=random_state,
            )
        ),
    ])

    # Wider search over C and gamma (log-scale-ish)
    param_grid = {
        "clf__C":    [0.3, 1, 3, 10, 30],
        "clf__gamma": ["scale", 1, 0.3, 0.1, 0.03, 0.01],
    }

    cv = StratifiedKFold(
        n_splits=5,
        shuffle=True,
        random_state=random_state,
    )

    grid = GridSearchCV(
        pipe,
        param_grid=param_grid,
        scoring="accuracy",
        cv=cv,
        n_jobs=-1,
        verbose=1,
    )

    print(f"Running GridSearch for {model_name}...")
    grid.fit(X_train, y_train)

    print("Best params:", grid.best_params_)
    print("Best CV accuracy:", grid.best_score_)

    best_clf = grid.best_estimator_

    # -------------------------
    # 4) Evaluation
    # -------------------------
    y_pred = best_clf.predict(X_test)
    acc = accuracy_score(y_test, y_pred)
    print(f"\n{model_name} — holdout accuracy: {acc:.3f}\n")

    print(
        classification_report(
            y_test,

            target_names=[int_to_class[i] for i in sorted(int_to_class.keys())],

    print("Confusion matrix [rows=true, cols=pred]:")

    joblib.dump((best_clf, class_to_int, int_to_class), out_path)
